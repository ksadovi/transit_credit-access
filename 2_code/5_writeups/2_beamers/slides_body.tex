\section{Overview}
\subsection{Motivation}
% ══════════════════════════════════════════════════════════════════
% ------ Slide 3 ------
\begin{frame}[label=motivation1]{A Historic Moment for Transit Investment}
\small
The Infrastructure Investment and Jobs Act (2021) commits \$91 billion to public
transit --- the largest federal investment in US transit history.
\bigskip
\begin{itemize}
  \item Capital Investment Grants program: \$23B in capital grants to rail and BRT projects nationwide.
  \item Dozens of new lines planned, under construction, or recently opened across the country.
  \item Urban congestion is rising and demand for accessible cities is growing.
\end{itemize}
\bigskip
Does physical access to a new rail station
actually improve labor-market outcomes for nearby residents?
\end{frame}
% ------ Slide 4 ------
\begin{frame}[label=mechanisms]{Mechanisms}
\small
A new rail station expands the reach of workers in three distinct but interconnected ways:
\bigskip
\begin{enumerate}
  \item \textbf{Job retention:} Reliable commuting makes it easier to hold existing jobs ---
        especially for workers without cars.
  \item \textbf{Job matching:} A larger effective job-search radius
        $\Rightarrow$ access to higher-paying or better-matched positions.
  \item \textbf{Networks and referrals:} Mobility expands the set of people, neighborhoods,
        and opportunities one can reach, broadening referral networks
        \parencite{patacchini_referral_fx}.
\end{enumerate}
\bigskip
My reduced-form question --- does gaining a station increase
employment and earnings for nearby residents?
\end{frame}
% ------ Slide 5 ------
\begin{frame}[label=heterogeneous]{Transit Impacts are Heterogeneous}
\small
Transit access is far from uniform:
\bigskip
\begin{itemize}
  \item Some stations extend lines into already-connected inner cities.
  \item Others bring first-ever rail access to historically transit-poor communities.
\end{itemize}
\bigskip
This matters both for identification and for interpretation:
\begin{itemize}
  \item A new station in a transit desert $\neq$ yet another station in Manhattan.
  \item Baseline network density shapes how much the new station expands the commute shed.
\end{itemize}
\bigskip
\textit{I return to this heterogeneity in the next steps
(slide~\ref{slide:nextsteps}).}
\end{frame}
% ------ Slide 6 ------
\begin{frame}[label=networks]{Why Networks? Why Now?}
\small
Labor markets are networked. Existing literature links:
\bigskip
\begin{itemize}
  \item Communication networks $\to$ job flows \parencite{patacchini_referral_fx}.
  \item Geographic dispersion $\to$ welfare losses from reduced social interaction
        \parencite{kimSpatialInteractions2023}.
  \item Proximity to financial services $\to$ credit access, especially for marginalized
        groups \parencite{Relihan2017BranchesIL}.
\end{itemize}
\bigskip
Transit is, in part, a \text{network expansion technology}. The question is whether
that expansion is large enough to register in labor-market outcomes --- and for whom.
\end{frame}

\subsection{Research Question \& Contribution}
% ══════════════════════════════════════════════════════════════════
% ------ Slide 7 ------
\begin{frame}[label=rq]{Research Question}
\begin{center}
\large
What is the causal effect of gaining access to a new rail station \\
on nearby residents' labor-market outcomes?
\end{center}
\bigskip\bigskip
\textbf{Outcomes:}
\begin{itemize}
    \item First order:
    \begin{itemize}
      \item Worker flows
    \end{itemize}
    \item Second order/next steps:
    \begin{itemize}
      \item Employment and earnings
      \item Credit access
      \item Migration
    \end{itemize}
\end{itemize}
\end{frame}
% ------ Slide 8 ------
\begin{frame}[label=contribution]{Contribution}
\small
\begin{enumerate}
  \item \textbf{New dataset:} Hand-coded projected and realized opening dates for stations
        across major US rail projects --- enabling a project-time delay measure unavailable
        in off-the-shelf data.
  \item \textbf{Identification:} Construction delays create quasi-random variation in
        \text{when} neighborhoods gain access, holding \text{which} neighborhoods were
        chosen constant. In progress: adapt \textcite{borusyakNonrandomExposureExogenous2023}
        recentering to a stacked DiD framework, purging non-random exposure from baseline
        geography.
  \item \textbf{Labor-market focus:} Unlike Tyndall's airport IV (city-specific geography),
        Baum-Snow's highway decentralization (metro-level spatial reorganization), and
        Tsivanidis's structural welfare model of Bogotá BRT, this paper uses delays ---
        endemic to every US rail project --- to isolate tract-level employment and earnings effects.
\end{enumerate}
\end{frame}

\section{Identification \& Model Design}
% ══════════════════════════════════════════════════════════════════
\subsection{Instruments \& Measurement Tools}
\begin{frame}[label=projtime]{Two Time Indexes: Project Time}
\textbf{Project time} $j$: where is this station in its planned lifecycle?
$$j = \text{census year} - \text{initial expected open year}$$
\begin{itemize}
  \item Defined for \text{all} stations --- opened, delayed, and never-opened alike
  \item $j < 0$: pre-expected-opening;\quad $j = 0$: originally scheduled opening year;\quad $j > 0$: overdue
  \item Fixed at the DEIS/FEIS stage, years before construction --- predates any local economic response
\end{itemize}
\end{frame}
\begin{frame}[label=eventtime]{Two Time Indexes: Event Time}
\textbf{Event time} $k$: how long has this station actually been open?
\begin{equation*}\label{eq:event_time}
  k = \text{census year} - \text{actual open year}
\end{equation*}
\begin{itemize}
  \item Defined \text{only} for stations that have opened ($k = \text{NA}$ for never-opened)
  \item $k = -1$ is the normalization period; $k \ge 0$ traces post-opening dynamics
\end{itemize}
\bigskip
\textit{Identification conditions on $j$; the event study traces effects in $k$.}
\end{frame}

\end{frame}
\subsection{Model Design}
\begin{frame}[label=twfe]{Static TWFE}
\[
  \log Y_{ij} = \alpha_i + \delta_j
  + \beta\,\mathbf{1}[\text{census year} \ge \text{open year}_i]
  + \mathbf{X}_{ij}'\gamma + \varepsilon_{ij}
\]
\begin{itemize}
  \item $i$: census tract;\quad $j =$ project-time
  \item $\alpha_i$: tract fixed effects;\quad $\delta_j$: project-time fixed effects
  \item $\mathbf{X}_{ij}'$: vector of tract-level controls. Median HH income, poverty rate, and station type.
  \item $\beta$: ATT of station opening on log worker inflows/outflows
\end{itemize}
\medskip
\textit{Problem:} actual open year may be endogenous to neighborhood conditions within project time. The next slide explains how the event study addresses this.
\end{frame}
\begin{frame}[label=eventstudy]{Event Study}
\[
  \log Y_{ij} = \delta_j
  + \sum_{k \ne -1}\sum_l \beta_{kl}
    \cdot \mathbf{1}[K_{ij} = k]
    \cdot \mathbf{1}[L_i = l]
  + \mathbf{X}_{ij}'\gamma + \varepsilon_{ij}
\]
\begin{itemize}
  \item $k = $: event time, normalized to $k = -1$
  \item $l \in \{5, 15, 30\}$: isochrone band (minutes of walking time)
  \item $\beta_{kl}$: dynamic ATT by event time and isochrone; $\beta_{kl}$ for $k \ge 0$ traces how the effect builds post-opening
  \item Control group: pre-opening observations from eventually-opened stations only (never-opened stations have $k = \text{NA}$ and are dropped)
  \item Flat pre-opening coefficients ($k < 0$) support parallel trends conditional on project-time $j$.
\end{itemize}
\end{frame}
\subsection{Identification Challenges}
\begin{frame}[label=challenges]{Three Identification Challenges}
\small
\begin{enumerate}
  \item \textbf{Anticipation effects:}
    \begin{itemize}
      \item Households and workers respond at announcement, not just at opening.
      \item House prices capitalize expected access up to 6 years early
            \parencite{mcmillenReactionHousePrices2004}.
      \item This is addressed by delay instrument.
    \end{itemize}
  \item \textbf{Non-random exposure:}
    \begin{itemize}
      \item Areas near new stations are already denser, more central, and often wealthier. This is addressed by isochrones.
    \end{itemize}
  \item \textbf{Endogenous delay length:}
    \begin{itemize}
      \item Longer delays may be correlated with neighborhood characteristics
            (opposition strength, wealth, fiscal stress).
    \end{itemize}
\end{enumerate}
\end{frame}
\begin{frame}[label=endogenous1]{Endogenous Delay Length}
\small
\textbf{Extensive margin is uniform} --- nearly every project has at least one
delay episode, so delay \text{status} is not selective across neighborhoods.
\bigskip
\textbf{The concern is the intensive margin:} Is delay \text{length} (or type)
correlated with neighborhood characteristics?
\end{frame}
\begin{frame}[label=endogenous2]{Endogenous Delay Length}
Possible sources of endogeneity in delay length:
\begin{itemize}
  \item \textbf{Neighborhood-level:}
  \begin{itemize}
      \item Political mobilization: Neighborhoods with more civic engagement may resist or accelerate projects
      \item Wealthier neighborhoods may have more political and legal resources to litigate or negotiate
  \end{itemize}
  \item \textbf{State-level:}
  \begin{itemize}
      \item Political characteristics: states with more political will to complete transit projects are different from those absent of it; same goes for other state-level characteristics like funding availability, bureaucratic density, etc.
      \item State-level funding capacity and federal grant timelines
  \end{itemize}
\end{itemize}
\end{frame}

\subsection{Challenges}
\begin{frame}[label=designchallenges]{Design challenges}
\small
Four anticipated challenges:
\begin{enumerate}
  \item \textbf{Neighborhood endogeneity:} Communities that oppose transit may differ
        systematically in socioeconomic outcomes --- the type of neighborhood
        that gets transit sooner may not be random.
  \item \textbf{City/state wealth heterogeneity:} Richer cities resolve delays faster;
        cross-city comparisons confound fiscal capacity with treatment timing.
  \item \textbf{Are delays truly exogenous?} Different delay types (political, funding,
        environmental) may be non-randomly distributed across neighborhoods.
  \item \textbf{Heterogeneous take-up:} Car-dependent cities (South) vs.\
        transit-oriented cities (Northeast); low-income vs.\ high-income users ---
        treatment effects may not generalize.
\end{enumerate}
\end{frame}
\subsection{Delay Instrument Validation}
\begin{frame}[label=test1arg]{Test 1: Delay Length Is Not Politically Structured}
\label{slide:political}
\small
\textbf{Political heterogeneity argument:}
\begin{itemize}
  \item MA and FL have similar delay distributions despite being politically
        polar opposites.
  \item MA, CA, NY, and VA are politically similar yet span the \text{full range}
        of delay lengths.
  \item HI's outlier status reflects HART's project-specific funding crisis ---
        a delay type also experienced by Miami.
\end{itemize}
\bigskip
$\Rightarrow$ Delay length reflects project characteristics,
not state-level political economy.

\section{Open Questions}
% ══════════════════════════════════════════════════════════════════
\begin{frame}[label=nextsteps]{Next Steps}
\label{slide:nextsteps}
\small
\textbf{Data \& measurement:}
\begin{itemize}
  \item Complete and finalize delay dataset across all systems
\end{itemize}
\bigskip
\textbf{Identification (in progress):}
\begin{itemize}
  \item Adapt \textcite{borusyakNonrandomExposureExogenous2023} recentering: construct
        $\tilde{z}_{ct} = z_{ct} - \mu_{ct}$ by permuting delays within
        project cohorts, purging non-random geographic exposure
  \item Embed in stacked DiD \parencite{wingStackedDifferenceinDifferences2024} with
        Wing--Freedman--Hollingsworth weights $\Rightarrow$ trimmed aggregate ATT
\end{itemize}
\bigskip
\textbf{Heterogeneous effects:}
\begin{itemize}
  \item Stratify by baseline transit access (transit-desert vs.\ connected),
        pedestrian network density, and neighborhood income
  \item By neighborhood income and transit take-up propensity
\end{itemize}
\end{frame}
% ------ Concern 1 ------
\begin{frame}[label=concern1]{Concern 1: Limited Delay Variation}
\small
\textbf{Within-system clumping:}
\begin{itemize}
  \item A system like BART or SFMTA contributes many stations, but may have only
        one or two system-wide delay episodes --- leaving little \textit{within-system}
        variation in delay length across stations.
\end{itemize}
\medskip
\textbf{Cross-system clumping:}
\begin{itemize}
  \item Sound Transit contributes 38 stations; Miami contributes just 1.
  \item Identifying variation may be dominated by a few large systems ---
        raising concerns about whether delay is truly a station-level shock.
\end{itemize}
\bigskip
$\Rightarrow$ Need to verify that sufficient within-system, cross-station
variation in delay length exists to estimate credible tract-level effects.
\end{frame}
% ------ Concern 2 ------
\begin{frame}[label=concern2]{Concern 2: SUTVA Violations and Spillovers}
\small
\textbf{Setup:} Residents near station B (not yet open) serve as controls.
But neighboring stations A and C have \textit{already opened}.

\medskip
\begin{center}
    \begin{tikzpicture}
        \draw[thick] (0,0) -- (9,0);
        \filldraw[green!60!black] (0,0) circle (8pt) node[below=8pt] {$A$};
        \filldraw[red]            (3,0) circle (8pt) node[below=8pt] {$B$};
        \filldraw[green!60!black] (6,0) circle (8pt) node[below=8pt] {$C$};
        \filldraw[black]          (9,0) circle (8pt) node[below=8pt] {$D$};
    \end{tikzpicture}
\end{center}

\medskip
B's residents can access newly-opened A and C $\Rightarrow$
``control'' units are partially treated.
Need to account for \textbf{treatment intensity} rather than binary treatment status.
\end{frame}
% ------ Concern 3 ------
\begin{frame}[label=concern3]{Concern 3: Incumbent vs.\ Incoming Residents}
\small
\textbf{The core identification problem:}
\begin{itemize}
  \item LODES measures workers \textit{at the tract}, not individuals over time.
  \item A station opening may change the \textit{composition} of who lives nearby ---
        attracting higher-income newcomers while displacing long-term residents.
  \item Measured ``gains'' in tract-level earnings or employment may reflect
        \textbf{population sorting}, not improvements for incumbent residents.
\end{itemize}
\bigskip
\textbf{What would fix it:}
\begin{itemize}
  \item Confidential LODES microdata or a commercial panel (e.g., Infogroup)
        to track the same individuals before and after station opening.
  \item Without this, the reduced-form estimates are composition effects as much
        as they are treatment effects on incumbents.
\end{itemize}
\end{frame}
