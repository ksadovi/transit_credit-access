\begin{frame}{Research Question, Motivation, and Approach}
\small

\textbf{Research question:} What is the causal effect of gaining access to a new rail station on local labor-market outcomes?

\medskip
\textbf{Why it matters:}
\begin{itemize}
  \item Transit changes effective access to jobs and people (commuting + networks), but station placement is endogenous.
  \item Many evaluations struggle to separate access effects from underlying trends in dense, already-connected places.
\end{itemize}
\medskip
\textbf{Broad strokes of the design:}
\begin{enumerate}
  \item \textbf{Measure access:} Use \texttt{r5r} to compute tract-level walking/transit travel times and a continuous market-access index (CMA).
  \item \textbf{Identify with timing:} Exploit quasi-random \emph{construction delays} (planned vs.\ realized opening dates) as shocks to when access improves.
  \item \textbf{Correct non-random exposure:} Apply Borusyak--Hull recentering to isolate delay-driven variation in access, and (in ongoing work) embed it in a stacked DiD event-time framework for dynamics.
\end{enumerate}

\end{frame}
