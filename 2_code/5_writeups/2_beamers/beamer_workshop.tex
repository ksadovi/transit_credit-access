\section{Motivation}
% ══════════════════════════════════════════════════════════════════

% ------ Slide 3 ------
\begin{frame}{A Historic Moment for Transit Investment}
\small
The Infrastructure Investment and Jobs Act (2021) commits \$91 billion to public transit --- the largest federal investment in US transit history.

\bigskip
\begin{itemize}
  \item New Starts program: \$23B in capital grants to rail and BRT projects nationwide.
  \item Dozens of new lines planned, under construction, or recently opened across the country.
\end{itemize}

\bigskip
\textbf{Yet we lack causal estimates:} Does physical access to a new rail station actually improve labor-market outcomes for nearby residents?

\bigskip
\textit{This paper provides that evidence.}
\end{frame}

% ------ Slide 4 ------
\begin{frame}{The Obvious Mechanism}
\small
A new rail station expands the reach of workers and firms simultaneously:

\bigskip
\begin{enumerate}
  \item \textbf{Job retention:} Reliable commuting makes it easier to hold existing jobs ---
        especially for workers without cars.
  \item \textbf{Job matching:} A larger effective job-search radius
        $\Rightarrow$ access to higher-paying or better-matched positions.
  \item \textbf{Networks and referrals:} Mobility expands the set of people, neighborhoods,
        and opportunities one can reach, broadening referral networks
        \parencite{patacchini_referral_fx}.
\end{enumerate}

\bigskip
Separating these channels is hard. I start with the reduced-form question: does gaining a station increase employment and earnings for nearby residents?
\end{frame}

% ------ Slide 5 ------
\begin{frame}{Geographic Friction Is Heterogeneous}
\small
Transit access is far from uniform:

\bigskip
\begin{itemize}
  \item Some stations extend lines into already-connected inner cities.
  \item Others bring \textbf{first-ever rail access} to historically transit-poor communities.
\end{itemize}

\bigskip
This matters both for identification and for interpretation:
\begin{itemize}
  \item A new station in a transit desert $\neq$ yet another station in Manhattan.
  \item Baseline network density shapes how much the new station expands the commute shed.
\end{itemize}

\bigskip
\textit{I return to this heterogeneity in the HTE discussion
(slide~\ref{slide:nextsteps}).}
\end{frame}

% ------ Slide 6 ------
\begin{frame}{Why Networks? Why Now?}
\small
Labor markets are networked. A tight body of evidence links:

\bigskip
\begin{itemize}
  \item Communication networks $\to$ job flows \parencite{patacchini_referral_fx}.
  \item Geographic dispersion $\to$ welfare losses from reduced social interaction
        \parencite{kim_spatial_2023}.
  \item Proximity to financial services $\to$ credit access, especially for marginalized groups
        \parencite{Relihan2017BranchesIL}.
\end{itemize}

\bigskip
Transit is, in part, a \emph{network expansion technology}. The question is whether that network expansion is large enough to register in labor-market outcomes --- and for whom.
\end{frame}

% ══════════════════════════════════════════════════════════════════
\section{Research Question \& Contribution}
% ══════════════════════════════════════════════════════════════════

% ------ Slide 7 ------
\begin{frame}{Research Question}
\begin{center}
\large
What is the causal effect of gaining access to a new rail station \\
on nearby residents' labor-market outcomes?
\end{center}

\bigskip\bigskip
\textbf{Outcomes:}
\begin{itemize}
  \item Employment and earnings (LODES)
  \item Worker flows: job retention, job-to-job transitions
\end{itemize}

\bigskip
\textbf{Key heterogeneity:}
\begin{itemize}
  \item By baseline transit access (transit desert vs.\ already connected)
  \item By neighborhood income and transit take-up propensity
\end{itemize}
\end{frame}

% ------ Slide 8 ------
\begin{frame}{Contribution}
\small
\begin{enumerate}
  \item \textbf{New dataset:} Hand-coded projected and realized opening dates for stations
        across major US rail projects --- enabling a project-time delay measure unavailable
        in off-the-shelf data.
  \item \textbf{Identification:} Construction delays create quasi-random variation in
        \emph{when} neighborhoods gain access, holding \emph{which} neighborhoods were
        chosen constant.
  \item \textbf{Instrument design:} Adapt \textcite{borusyak_nonrandom_2023} recentering
        to a stacked DiD framework, purging non-random exposure from baseline geography.
  \item \textbf{HTE:} First estimates stratified by baseline transit access --- distinguishing
        transit-desert gains from incremental additions to already-connected networks.
\end{enumerate}
\end{frame}

% ══════════════════════════════════════════════════════════════════
\section{Identification Design}
% ══════════════════════════════════════════════════════════════════

% ------ Slide 9 ------
\begin{frame}{The Basic Estimating Equation}
\small
\textbf{Baseline TWFE:}
\[
  y_{i,j} = \alpha_i + \alpha_j
           + \beta \cdot \text{Open}_{s(i),j} + \varepsilon_{i,j}
\]
\begin{itemize}
  \item $j = t - P_s$: project-time index (0 = projected opening for station $s$)
  \item $\text{Open}_{s(i),j} = \mathbf{1}\!\left\{j \ge D_{s(i)}\right\}$:
        station is open by project-time $j$
  \item $\alpha_i$: unit FE;\quad $\alpha_j$: project-time FE
\end{itemize}

\medskip
\textbf{Event-study extension:}
\[
  y_{i,j} = \alpha_i + \alpha_j
  + \sum_{k \ne -1} \beta_k \cdot
    \mathbf{1}\!\left\{j = D_{s(i)} + k\right\} + \varepsilon_{i,j}
\]
Traces the dynamic ATT around the actual opening date.

\medskip
\textit{Problem:} $D_{s(i)}$ is the delay, and delay length may be endogenous.
The next slides explain why and how I address it.
\end{frame}

% ------ Slide 10 ------
\begin{frame}{Three Identification Challenges}
\small
\begin{enumerate}
  \item \textbf{Anticipation effects:}
    \begin{itemize}
      \item Households and firms respond at announcement, not just at opening.
      \item House prices capitalize expected access up to 6 years early
            \parencite{mcmillen_reaction_2004}.
    \end{itemize}
  \item \textbf{Non-random exposure:}
    \begin{itemize}
      \item Areas near new stations are already denser, more central, and often wealthier.
      \item Standard DiD absorbs station FEs but not the spatial gradient around them.
    \end{itemize}
  \item \textbf{Endogenous delay length:}
    \begin{itemize}
      \item Longer delays may be correlated with neighborhood characteristics
            (opposition strength, wealth, fiscal stress).
      \item If delay length predicts baseline outcomes, using $D_s$ directly
            as variation is invalid.
    \end{itemize}
\end{enumerate}
\end{frame}

% ------ Slide 11 ------
\begin{frame}{Delays Provide the Timing Variation}
\small
\textbf{Core idea:} Two neighborhoods near the same line are both at project-time $j=0$
on the same calendar date. One station opens on schedule; the other is delayed.
The delayed neighborhood acts as a within-project-time control.

\bigskip
\textbf{Formally:} Index time by $j = t - P_s$ so all projects share a common
origin $j=0$. Variation in $D_s$ shifts when $\text{Open}_{s,j}$ flips from~0~to~1.

\bigskip
\textbf{Requires:} Delay length $D_s$ is ``as good as random'' conditional on
baseline geography and project characteristics.

\medskip
\textit{But is it? I test this directly on slides
\ref{slide:political}--\ref{slide:withinsystem}.}
\end{frame}

% ------ Slide 12 ------
\begin{frame}{A Taxonomy of Delays}
\small
Nearly every project experiences delay --- the \textbf{extensive margin} is
fairly uniform. The variation I exploit is in \textbf{delay length}
(the intensive margin).

\medskip
Three recurring causes, with two examples each:

\begin{enumerate}
  \item \textbf{Funding freezes / cost overruns}
    \begin{itemize}
      \item \textit{Honolulu (HART):} FTA flagged a projected \$1B overrun and 5-year
            delay, threatening to withhold federal funding mid-construction
            \parencite{HART_FTA_2016}.
      \item \textit{NYC East Side Access (MTA):} Budget grew from \$6.3B to over \$11B;
            opening slipped from 2013 to 2023 \parencite{NYT_ESA_2014}.
    \end{itemize}
  \item \textbf{Neighborhood / political opposition}
    \begin{itemize}
      \item \textit{LA Purple Line:} Beverly Hills USD sued to block tunneling under
            Beverly Hills High School, stalling the extension for years.
      \item \textit{MD Purple Line:} Georgetown Branch trail advocates contested
            right-of-way since 1983; NEPA scoping began 2003, FEIS issued August 2013.
    \end{itemize}
  \item \textbf{Environmental review}
    \begin{itemize}
      \item \textit{LA Purple Line:} Five-year environmental review (2007--2012)
            before the ROD was issued.
      \item \textit{HART (mid-construction):} Supplemental EIS and
            Section~4(f) evaluation required; completed September 2013.
    \end{itemize}
\end{enumerate}
\end{frame}

% ------ Slide 13 ------
\begin{frame}{But Delays Alone Aren't Enough}
\small
\textbf{Extensive margin is uniform} --- nearly every project has at least one
delay episode, so delay \emph{status} is not selective across neighborhoods.

\bigskip
\textbf{The concern is the intensive margin:} Is delay \emph{length} correlated
with neighborhood characteristics?

\bigskip
Possible sources of endogeneity in delay length:
\begin{itemize}
  \item Wealthier neighborhoods may have more political and legal resources
        to litigate or negotiate
  \item Fiscally constrained cities may resolve funding gaps more slowly
  \item \textbf{Political mobilization:} neighborhoods with more civic engagement
        may resist or accelerate projects
  \item State-level funding capacity and federal grant timelines
\end{itemize}

\bigskip
$\Rightarrow$ I need to show delay length is not systematically predicted by
neighborhood characteristics \emph{within} a transit system. I do this next.
\end{frame}

% ------ Slide 14 ------
\begin{frame}{Borusyak--Hull Recentering: Background}
\small
\textcite{borusyak_nonrandom_2023} identify a general problem in shift-share designs:
the exposure measure $z_{ct}(g_t; w)$ inherits non-randomness from baseline geography
$w$, even when shocks $g_t$ are random.

\bigskip
\textbf{Solution:} Construct the \emph{recentered} instrument
\[
  \tilde{z}_{ct} = z_{ct}(g_t; w) - \underbrace{\mu_{ct}}_{\text{expected exposure}},
  \qquad
  \mu_{ct} = \frac{1}{B}\sum_{b=1}^{B} z_{ct}\!\left(g_t^{(b)}; w\right),
\]
where $\bigl\{g_t^{(b)}\bigr\}$ are counterfactual shock draws from the
distribution $G(g \mid w)$.

\bigskip
\textbf{Key condition:} Shocks $g_t$ are ``as good as random'' --- i.e., the realized
delay vector is exchangeable with permuted delay vectors within cohorts of
similar projects.
\end{frame}

% ------ Slide 15 ------
\begin{frame}{Adapting BH to This Setting}
\small
\textbf{Shocks:} Station-level delays $D_s = R_s - P_s$.
Shock vector $g = \{D_s\}_s$.

\bigskip
\textbf{Exposure:} Walking-time-weighted access to open stations:
\[
  z_{ct}(g; w) = \sum_{s \in \mathcal{S}_c(15)}
    \text{Open}_{s,t}(g) \cdot w_{cs},
  \quad
  w_{cs} =
    \mathbf{1}[d_{cs} \!\in\! \mathcal{B}_1]
    + \tfrac{2}{3}\mathbf{1}[d_{cs} \!\in\! \mathcal{B}_2]
    + \tfrac{1}{3}\mathbf{1}[d_{cs} \!\in\! \mathcal{B}_3]
\]
with bands $\mathcal{B}_1 = [0,5]$, $\mathcal{B}_2 = (5,10]$,
$\mathcal{B}_3 = (10,15]$ minutes walk.

\bigskip
\textbf{Counterfactuals:} Permute $\{D_s\}$ within cohorts (by line type and
construction phase), holding planned dates $P_s$ fixed.

\bigskip
\textbf{Result:} $\tilde{z}_{ct}$ retains the geography-induced level of
access $\mu_{ct}$, leaving only quasi-random delay-induced variation.
Use in 2SLS or stacked DiD.
\end{frame}

% ------ Slide 16 ------
\begin{frame}{Putting It Together: Stacked DiD + BH}
\small
Embed $\tilde{z}_{ct}$ in a stacked DiD framework \parencite{wing_stacked_2024}:

\begin{itemize}
  \item For each planned opening year $a$, form a sub-experiment:
    \begin{itemize}
      \item \textit{Treated:} tracts whose nearest station is scheduled in year $a$
      \item \textit{Controls:} tracts not yet treated in the event window
    \end{itemize}
  \item Stack sub-experiments; apply Wing--Freedman--Hollingsworth weights
        $\Rightarrow$ trimmed aggregate ATT.
  \item Event-time coefficients trace the dynamic effect
        relative to projected opening $j = 0$.
\end{itemize}

\bigskip
\textbf{Two uses of $\tilde{z}_{ct}$:}
\begin{enumerate}
  \item As the treatment itself: effect of an exogenous access gain
  \item As an instrument for CMA (commuter market access),
        recovering a continuous dose--response
\end{enumerate}
\end{frame}

\section{Validation}
% ══════════════════════════════════════════════════════════════════

% ------ Slide 19 ------
\begin{frame}{Concerns I'm Addressing Head-On}
\small
Four anticipated challenges:

\begin{enumerate}
  \item \textbf{Neighborhood endogeneity:} Communities that oppose transit may differ
        systematically in socioeconomic outcomes --- i.e., the type of neighborhood
        that gets transit (or gets it sooner) may not be random.
  \item \textbf{City/state wealth heterogeneity:} Richer cities resolve delays faster;
        cross-city comparisons confound fiscal capacity with treatment timing.
  \item \textbf{Are delays truly exogenous?} Different delay types (political, funding,
        environmental) may be non-randomly distributed across neighborhoods.
  \item \textbf{Heterogeneous take-up:} Car-dependent cities (South) vs.\
        transit-oriented cities (Northeast); low-income vs.\ high-income users ---
        treatment effects may not generalize.
\end{enumerate}

\bigskip
\textit{Next three slides address concerns 1--3 directly.}
\end{frame}

\section{Empirical Specification}
% ══════════════════════════════════════════════════════════════════

% ------ Slide 23 ------
\begin{frame}{Main Specification}
\small
\textbf{Reduced form:}
\[
  y_{c,j} = \alpha_c + \alpha_j
  + \beta\,\tilde{z}_{c,j} + \Gamma' V_c + \varepsilon_{c,j}
\]
\begin{itemize}
  \item $y_{c,j}$: labor-market outcome in tract $c$ at project-time $j$
  \item $\tilde{z}_{c,j}$: recentered walking-time-weighted access to open stations
  \item $\alpha_c$, $\alpha_j$: tract and project-time fixed effects
  \item $V_c$: baseline tract controls (income, population, employment rate)
\end{itemize}

\bigskip
\textbf{2SLS:} Instrument continuous CMA with $\tilde{z}_{c,j}$;
second stage recovers dose--response.

\bigskip
\textbf{HTE:} Interact $\tilde{z}_{c,j}$ with:
\begin{itemize}
  \item Baseline transit-network density (transit-desert indicator)
  \item Pedestrian network density
  \item Neighborhood income quintile
\end{itemize}
\end{frame}

% ------ Slide 24 ------
\begin{frame}{Event-Study Specification}
\small
\textbf{Dynamic ATT:}
\[
  y_{c,j} = \alpha_c + \alpha_j
  + \sum_{k \ne -1} \beta_k\,\tilde{z}_{c,\,j=k}
  + \Gamma' V_c + \varepsilon_{c,j}
\]
\begin{itemize}
  \item $k$: event time relative to projected opening ($k=-1$ excluded)
  \item Pre-trend coefficients ($k < 0$) test for anticipation effects
  \item Post-opening coefficients ($k \ge 0$) trace the dynamic labor-market response
\end{itemize}

\bigskip
Embed in stacked DiD \parencite{wing_stacked_2024} to recover a trimmed
aggregate $\text{ATT}(a, a+k)$ across opening cohorts.
\end{frame}

\section{Open Questions}
% ══════════════════════════════════════════════════════════════════

% ------ Slide 25 ------
\begin{frame}{What I'm Wrestling With}
\small
\begin{itemize}
  \item \textbf{Control group:} Clean controls vs.\ never-treated vs.\
        not-yet-treated in the stacked DiD --- different choices yield
        different estimands.
  \item \textbf{Anticipation:} Households start responding at announcement,
        not at $j=0$. Should the ``event'' shift to the announcement date?
        How do I estimate the announcement effect itself?
  \item \textbf{Delay classification:} Cleanly assigning funding / opposition /
        environmental type to each delay episode; whether delay type should
        enter the BH permutation cohorts.
  \item \textbf{Stations that move:} A handful of stations change location during
        environmental review --- they should be excluded from the exposure measure.
  \item \textbf{Take-up and HTE:} Proxying individual-level transit propensity
        without individual data; car-ownership rates at the tract level
        are a starting point.
\end{itemize}
\end{frame}

% ------ Slide 26 ------
\begin{frame}{Next Steps}
\label{slide:nextsteps}
\small
\textbf{Data \& measurement:}
\begin{itemize}
  \item Complete delay dataset for remaining systems
  \item Automated delay-type classification from EIS narrative text (NLP)
  \item Recentered instrument $\tilde{z}_{ct}$ for all tract--station pairs
\end{itemize}

\bigskip
\textbf{Estimation:}
\begin{itemize}
  \item Baseline TWFE with recentered instrument
  \item Stacked DiD event study with Wing--Freedman--Hollingsworth weights
  \item HTE stratified by: (i) baseline transit access
        (transit-desert vs.\ already-connected neighborhoods),
        (ii) pedestrian network density, (iii) neighborhood income quintile
\end{itemize}

\bigskip
\textbf{Questions for today:}
\begin{enumerate}
  \item Is the intensive- vs.\ extensive-margin framing for delays convincing?
  \item How should I define the control group in the stacked DiD?
  \item Are there other sources of exogenous delay variation I'm missing?
  \item Should take-up heterogeneity be a primary result or a robustness check?
\end{enumerate}
\end{frame}
