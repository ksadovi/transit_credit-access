\section{Motivation}
% ══════════════════════════════════════════════════════════════════

% ------ Slide 3 ------
\begin{frame}{A Historic Moment for Transit Investment}
\small
The Infrastructure Investment and Jobs Act (2021) commits \$91 billion to public
transit --- the largest federal investment in US transit history.

\bigskip
\begin{itemize}
  \item New Starts program: \$23B in capital grants to rail and BRT projects nationwide.
  \item Dozens of new lines planned, under construction, or recently opened across the country.
\end{itemize}

\bigskip
Urban congestion is rising and demand for accessible cities is growing ---
yet it is unclear whether transit investment actually delivers for workers.

\bigskip
\textbf{Yet we lack causal estimates:} Does physical access to a new rail station
actually improve labor-market outcomes for nearby residents?
\end{frame}

% ------ Slide 4 ------
\begin{frame}{The Obvious Mechanism}
\small
A new rail station expands the reach of workers in three distinct but interconnected ways:

\bigskip
\begin{enumerate}
  \item \textbf{Job retention:} Reliable commuting makes it easier to hold existing jobs ---
        especially for workers without cars.
  \item \textbf{Job matching:} A larger effective job-search radius
        $\Rightarrow$ access to higher-paying or better-matched positions.
  \item \textbf{Networks and referrals:} Mobility expands the set of people, neighborhoods,
        and opportunities one can reach, broadening referral networks
        \parencite{patacchini_referral_fx}.
        % TODO: add Patacchini referral-networks citation key once confirmed
\end{enumerate}

\bigskip
My focus: the reduced-form question --- does gaining a station increase
employment and earnings for nearby residents?
\end{frame}

% ------ Slide 5 ------
\begin{frame}{Geographic Friction Is Heterogeneous}
\small
Transit access is far from uniform:

\bigskip
\begin{itemize}
  \item Some stations extend lines into already-connected inner cities.
  \item Others bring first-ever rail access to historically transit-poor communities.
\end{itemize}

\bigskip
This matters both for identification and for interpretation:
\begin{itemize}
  \item A new station in a transit desert $\neq$ yet another station in Manhattan.
  \item Baseline network density shapes how much the new station expands the commute shed.
\end{itemize}

\bigskip
\textit{I return to this heterogeneity in the next steps
(slide~\ref{slide:nextsteps}).}
\end{frame}

% ------ Slide 6 ------
\begin{frame}{Why Networks? Why Now?}
\small
Labor markets are networked. Existing literature links:

\bigskip
\begin{itemize}
  \item Communication networks $\to$ job flows \parencite{patacchini_referral_fx}.
  \item Geographic dispersion $\to$ welfare losses from reduced social interaction
        \parencite{kim_spatial_2023}.
  \item Proximity to financial services $\to$ credit access, especially for marginalized
        groups \parencite{Relihan2017BranchesIL}.
\end{itemize}

\bigskip
Transit is, in part, a \emph{network expansion technology}. The question is whether
that expansion is large enough to register in labor-market outcomes --- and for whom.
\end{frame}

% ══════════════════════════════════════════════════════════════════
\section{Research Question \& Contribution}
% ══════════════════════════════════════════════════════════════════

% ------ Slide 7 ------
\begin{frame}{Research Question}
\begin{center}
\large
What is the causal effect of gaining access to a new rail station \\
on nearby residents' labor-market outcomes?
\end{center}

\bigskip\bigskip
\textbf{Outcomes:}
\begin{itemize}
  \item Employment and earnings (LODES)
  \item Worker flows: job retention, job-to-job transitions
\end{itemize}

\end{frame}

% ------ Slide 8 ------
\begin{frame}{Contribution}
\small
\begin{enumerate}
  \item \textbf{New dataset:} Hand-coded projected and realized opening dates for stations
        across major US rail projects --- enabling a project-time delay measure unavailable
        in off-the-shelf data.
  \item \textbf{Identification:} Construction delays create quasi-random variation in
        \emph{when} neighborhoods gain access, holding \emph{which} neighborhoods were
        chosen constant.
  \item \textbf{Instrument design (in progress):} Adapt \textcite{borusyak_nonrandom_2023}
        recentering to a stacked DiD framework, purging non-random exposure from baseline
        geography.
  \item \textbf{Labor-market focus:} Unlike Tyndall's airport IV (city-specific geography),
        Baum-Snow's highway decentralization (metro-level spatial reorganization), and
        Tsivanidis's structural welfare model of Bogotá BRT, this paper uses delays ---
        endemic to every US rail project --- to isolate tract-level employment and earnings effects.
\end{enumerate}
\end{frame}

\section{Identification Design}
% ══════════════════════════════════════════════════════════════════

% ------ Slide 10 ------
\begin{frame}{The Basic Estimating Equation}
\small
\textbf{Baseline TWFE:}
\[
  y_{i,j} = \alpha_i + \alpha_j
           + \beta \cdot \text{Open}_{s(i),j} + \varepsilon_{i,j}
\]
\begin{itemize}
  \item $j = t - P_s$: project-time index (0 = projected opening for station $s$)
  \item $\text{Open}_{s(i),j} = \mathbf{1}\!\left\{j \ge D_{s(i)}\right\}$:
        station is open by project-time $j$
  \item $\alpha_i$: census tract FE;\quad $\alpha_j$: project-time FE
\end{itemize}

\medskip
\textbf{Event-study extension:}
\[
  y_{i,j} = \alpha_i + \alpha_j
  + \sum_{k \ne -1} \beta_k \cdot
    \mathbf{1}\!\left\{j = D_{s(i)} + k\right\} + \varepsilon_{i,j}
\]
Traces the dynamic ATT around the actual opening date.

\medskip
\textit{Problem:} $D_{s(i)}$ is the delay --- delay length may be endogenous.
The next slides explain why and how I address it.
\end{frame}

% ------ Slide 10 ------
\begin{frame}{Three Identification Challenges}
\small
\begin{enumerate}
  \item \textbf{Anticipation effects:}
    \begin{itemize}
      \item Households and workers respond at announcement, not just at opening.
      \item House prices capitalize expected access up to 6 years early
            \parencite{mcmillen_reaction_2004}.
    \end{itemize}
  \item \textbf{Non-random exposure:}
    \begin{itemize}
      \item Areas near new stations are already denser, more central, and often wealthier.
    \end{itemize}
  \item \textbf{Endogenous delay length:}
    \begin{itemize}
      \item Longer delays may be correlated with neighborhood characteristics
            (opposition strength, wealth, fiscal stress).
    \end{itemize}
\end{enumerate}
\end{frame}

% ------ Slide 11 ------
\begin{frame}{Delays Provide the Timing Variation}
\small
\textbf{Core idea:} Two neighborhoods near the same line are both at project-time $j=0$
on the same calendar date. One station opens on schedule; the other is delayed.
The delayed neighborhood acts as a within-project-time control.

\bigskip
\textbf{Formally:} Index time by $j = t - P_s$ so all projects share a common
origin $j=0$. Variation in $D_s$ shifts when $\text{Open}_{s,j}$ flips from 0 to 1.

\bigskip
\textbf{Requires:} Delay length $D_s$ is ``as good as random'' conditional on
baseline geography and neighborhood characteristics.

\medskip
\textit{But is it? I test this directly on slides
\ref{slide:political}--\ref{slide:crosssystem}.}
\end{frame}

% ------ Slide 13 ------
\begin{frame}{But Delays Alone Aren't Enough}
\small
\textbf{Extensive margin is uniform} --- nearly every project has at least one
delay episode, so delay \emph{status} is not selective across neighborhoods.

\bigskip
\textbf{The concern is the intensive margin:} Is delay \emph{length} (or type)
correlated with neighborhood characteristics?

\bigskip
Possible sources of endogeneity in delay length:
\begin{itemize}
  \item Wealthier neighborhoods may have more political and legal resources
        to litigate or negotiate
  \item Fiscally constrained cities may resolve funding gaps more slowly
  \item \textbf{Political mobilization:} neighborhoods with more civic engagement
        may resist or accelerate projects
  \item State-level funding capacity and federal grant timelines
\end{itemize}

\end{frame}

\begin{frame}{Constructing the Project-Time Index}
\small
All analyses are indexed by $j = t - P_s$ (years from projected opening):

\bigskip

\def\xmin{0}
\pgfmathsetmacro{\xmid}{(\xmin + \xmax)/2}
\begin{tikzpicture}
  % Top arrow: delayed station
  \draw[dashed, ->, thick] (\xmin,1) -- (\xmax,1);
  \node at (\xmin,1) {\textbullet};
  \node at (\xmid,1) {\textbullet};
  \node[above] at (\xmin,1) {\footnotesize Announced};
  \node[above] at (\xmid,1) {\footnotesize Projected open ($j=0$)};
  \node[right] at (\xmax,1) {\footnotesize $\text{Open}=0$ \textit{(delayed)}};
  % Bottom arrow: on-time station
  \draw[dashed, thick] (\xmin,0) -- (\xmid,0);
  \draw[->, thick, uchicagored] (\xmid,0) -- (\xmax,0);
  \node at (\xmin,0) {\textbullet};
  \node at (\xmid,0) {\textbullet};
  \node[below] at (\xmin,0) {\footnotesize $j < 0$};
  \node[below] at (\xmid,0) {\footnotesize $j=0$};
  \node[right] at (\xmax,0) {\footnotesize $\text{Open}=1$ \textit{(on-time)}};
\end{tikzpicture}

\bigskip

those near still-delayed stations. Delays generate as-good-as-random variation
in $\text{Open}_{s,j}$ at each project-time.
\end{frame}

\section{Validation}
% ══════════════════════════════════════════════════════════════════

% ------ Slide 16 ------
\begin{frame}{Concerns I'm Addressing Head-On}
\small
Four anticipated challenges:

\begin{enumerate}
  \item \textbf{Neighborhood endogeneity:} Communities that oppose transit may differ
        systematically in socioeconomic outcomes --- the type of neighborhood
        that gets transit sooner may not be random.
  \item \textbf{City/state wealth heterogeneity:} Richer cities resolve delays faster;
        cross-city comparisons confound fiscal capacity with treatment timing.
  \item \textbf{Are delays truly exogenous?} Different delay types (political, funding,
        environmental) may be non-randomly distributed across neighborhoods.
  \item \textbf{Heterogeneous take-up:} Car-dependent cities (South) vs.\
        transit-oriented cities (Northeast); low-income vs.\ high-income users ---
        treatment effects may not generalize.
\end{enumerate}

\bigskip
\textit{Next four slides address concerns 1--3 directly.}
\end{frame}

% ------ Slide 17 (Political: argument) ------
\begin{frame}{Delay Length Is Not Politically Structured}
\label{slide:political}
\small
\textbf{Political heterogeneity argument:}
\begin{itemize}
  \item MA and FL have similar delay distributions despite being politically
        polar opposites.
  \item MA, CA, NY, and VA are politically similar yet span the \emph{full range}
        of delay lengths.
  \item HI's outlier status reflects HART's project-specific funding crisis ---
        a delay type also experienced by Miami.
\end{itemize}

\bigskip
$\Rightarrow$ Delay length reflects project characteristics,
not state-level political economy.

\bigskip

\end{frame}

\section{Open Questions}
% ══════════════════════════════════════════════════════════════════

% ------ Slide 21 ------
\begin{frame}{What I'm Wrestling With}
\small
\begin{itemize}
  \item \textbf{Control group:} Clean controls vs.\ never-treated vs.\
        not-yet-treated in the stacked DiD --- different choices yield
        different estimands.
  \item \textbf{Anticipation:} Households start responding at announcement,
        not at $j=0$. Should the ``event'' shift to the announcement date?
        How do I estimate the announcement effect itself?
  \item \textbf{Delay classification:} Cleanly assigning funding / opposition /
        environmental type to each delay episode; and whether delay type should
        enter the BH permutation cohorts.
  \item \textbf{Stations that move:} A handful of stations change location during
        environmental review --- they should be excluded from the exposure measure.
  \item \textbf{Take-up and HTE:} Proxying individual-level transit propensity
        without individual data; car-ownership rates at the tract level
        are a starting point.
\end{itemize}
\end{frame}

% ------ Slide 22 ------
\begin{frame}{Next Steps}
\label{slide:nextsteps}
\small
\textbf{Data \& measurement:}
\begin{itemize}
  \item Complete and finalize delay dataset across all systems
  \item Automated delay-type classification from EIS narrative text (NLP)
\end{itemize}

\bigskip
\textbf{Identification (in progress):}
\begin{itemize}
  \item Adapt \textcite{borusyak_nonrandom_2023} recentering: construct
        $\tilde{z}_{ct} = z_{ct} - \mu_{ct}$ by permuting delays within
        project cohorts, purging non-random geographic exposure
  \item Embed in stacked DiD \parencite{wing_stacked_2024} with
        Wing--Freedman--Hollingsworth weights $\Rightarrow$ trimmed aggregate ATT
\end{itemize}

\bigskip
\textbf{Heterogeneous effects:}
\begin{itemize}
  \item Stratify by baseline transit access (transit-desert vs.\ connected),
        pedestrian network density, and neighborhood income
  \item By neighborhood income and transit take-up propensity
\end{itemize}
\end{frame}

% ------ Questions for today (new slide) ------
\begin{frame}{Questions for Today}
\begin{enumerate}
  \item Is the intensive- vs.\ extensive-margin framing for delays convincing?
  \item How should I define the control group in the stacked DiD?
  \item Are there other sources of exogenous delay variation I'm missing?
  \item Should take-up heterogeneity be a primary result or a robustness check?
\end{enumerate}
\end{frame}
